#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include "gbv.h"

// NOVO: Defina a sua chave de 4 caracteres aqui
#define CHAVE_MESTRA "1234" 

int main(int argc, char *argv[]) {

    // MUDOU: O número mínimo de argumentos sobe para 4 (programa, chave, opcao, biblioteca)
    if (argc < 4) {
        printf("Uso: %s <chave> <opção> <biblioteca> [documentos...]\n", argv[0]);
        return 1;
    }

    // NOVO: Início da validação da chave de segurança
    char *chave_fornecida = argv[1];

    if (strlen(chave_fornecida) != 4) {
        printf("Erro: A chave de seguranca deve ter exatamente 4 caracteres.\n");
        return 1;
    }

    if (strcmp(chave_fornecida, CHAVE_MESTRA) != 0) {
        printf("Acesso Negado: Chave de seguranca incorreta.\n");
        return 1;
    }
    // NOVO: Fim da validação da chave

    // MUDOU: Como a chave ocupou argv[1], os próximos argumentos andam uma casa
    const char *opcao = argv[2];      // Antes era argv[1]
    const char *biblioteca = argv[3]; // Antes era argv[2]

    Library lib;

    // Alteracao: Criar a biblioteca se for inserção e ela não existir 
    FILE *teste_existe;
    teste_existe = fopen(biblioteca, "rb");

    if (!teste_existe && strcmp(opcao, "-a") == 0) 
        gbv_create(biblioteca);

    else if (teste_existe)
        fclose(teste_existe);

    if (gbv_open(&lib, biblioteca) != 0) {
        printf("Erro ao abrir biblioteca %s\n", biblioteca);
        return 1;
    }

    if (strcmp(opcao, "-a") == 0) {
        // MUDOU: O loop agora começa em 4, pois os documentos começam em argv[4]
        for (int i = 4; i < argc; i++) {
            gbv_add(&lib, biblioteca, argv[i]);
        }
    } 

    else if (strcmp(opcao, "-r") == 0) {
        // MUDOU: O loop agora começa em 4
        for (int i = 4; i < argc; i++) {
            gbv_remove(&lib, biblioteca, argv[i]);
        }
    } 

    else if (strcmp(opcao, "-l") == 0) {
        gbv_list(&lib);
    } 

    // MUDOU: argc >= 5 e argv[4] (andou uma casa)
    else if (strcmp(opcao, "-v") == 0 && argc >= 5) {
        gbv_view(&lib, biblioteca, argv[4]);// alteração realizada para receber
                                            // paramentro 
    } 

    // MUDOU: argc >= 5 e argv[4] (andou uma casa)
    else if (strcmp(opcao, "-o") == 0 && argc >= 5) {
        gbv_order(&lib, biblioteca, argv[4]);
    } 

    else {
        printf("Opção inválida ou argumentos insuficientes.\n");
    }

    // alteracao: Liberar a memória do diretório antes de encerrar o programa
    if (lib.docs != NULL) {
        free(lib.docs);
        lib.docs = NULL;
    }

    return 0;
}
